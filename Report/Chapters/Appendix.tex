\begin{lstlisting}[
  language=Lean4,
  caption={Common typeclasses and notation used throughout the Lean development. These abbreviations are used uniformly for shifting, substitution, opening, closing, prefix notation, and permutation.},
  label={lst:lean-common-notation}
]
class HasShift (Subject : Type) where
  shift : Subject -> Nat -> Nat -> Subject

notation:max S:max " /-$\uparrow$-/ " d:max ", " c:max => HasShift.shift S d c
notation:max S:max " /-$\uparrow$-/ " c:max => HasShift.shift S 0 c
notation:max S:max "/-$^+$-/" => HasShift.shift S 0 1

class HasSubst (Subject Replacement Target : Type) where
  subst : Subject -> Replacement -> Target -> Subject

notation:max S "{" R " // " T "}" => HasSubst.subst S R T

class HasOpen (Subject Replacement Target : Type) where
  open_ : Subject -> Replacement -> Target -> Subject

notation:max S:max "/-$\dparenL$-/" T " | " R "/-$\dparenR$-/" => HasOpen.open_ S R T
notation:max S:max "/-$\dparenL$-/" R "/-$\dparenR$-/" => HasOpen.open_ S R 0

class HasClose (Subject Replacement Target : Type) where
  close_ : Subject -> Replacement -> Target -> Subject

notation:max S:max "/-$\dangleL$-/" T " | " R "/-$\dangleR$-/" => HasClose.close_ S R T
notation:max S:max "/-$\dangleL$-/" R "/-$\dangleR$-/" => HasClose.close_ S R 0

class HasOpenTwo (Subject Replacement_1 Replacement_2 Target : Type) where
  open_ : Subject -> Replacement_1 -> Replacement_2 -> Target -> Subject

notation:max S:max "/-$\dparenL$-/" T " | " R1 ", " R2 "/-$\dparenR$-/" => HasOpenTwo.open_ S R1 R2 T
notation:max S:max "/-$\dparenL$-/" R1 ", " R2 "/-$\dparenR$-/" => HasOpenTwo.open_ S R1 R2 0

class HasCloseTwo (Subject Replacement_1 Replacement_2 Target : Type) where
  close_ : Subject -> Replacement_1 -> Replacement_2 -> Target -> Subject

notation:max S:max "/-$\dangleL$-/" T " | " R1 ", " R2 "/-$\dangleR$-/" => HasCloseTwo.close_ S R1 R2 T
notation:max S:max "/-$\dangleL$-/" R1 ", " R2 "/-$\dangleR$-/" => HasCloseTwo.close_ S R1 R2 0

class HasBracket (Subject Content Result : Type) where
   brack : Subject -> Content -> Result

class HasParen (Subject Content Result : Type) where
   paren : Subject -> Content -> Result

notation:80 x"/-$\dbrackL\dbrackR$-/" => HasBracket.brack x ()
notation:80 x"/-$\dbrackL$-/"y"/-$\dbrackR$-/" => HasBracket.brack x y
notation:80 x"/-$\dparenL\dparenR$-/" => HasParen.paren x ()
notation:80 x"/-$\dparenL$-/ y/-$\dparenR$-/" => HasParen.paren x y

class HasPerm (/-$\alpha$-/ : Type) where
  perm : /-$\alpha$-/ -> /-$\alpha$-/ -> Prop

infixr:54 " /-$\sim$-/ " => HasPerm.perm

\subsection{Processes}\label{app:processes}
This appendix gives the core Lean representation of channels and process syntax. Channels are represented using the locally nameless approach discussed in \cref{sec:binding-architecture}, separating free names from bound indices.

\begin{lstlisting}[
  language=Lean4,
  caption={Locally nameless representation of channels and process syntax. Free channels are represented explicitly, while bound channels are represented by De Bruijn indices.},
  label={lst:lean-proc-basic}
]
abbrev FPName := Nat
abbrev BPName := Nat

inductive Channel : Type where
  | free    (x : FPName)
  | bound   (x : BPName)
deriving DecidableEq, BEq, Repr

prefix:max "#" => Channel.free
prefix:max "$" => Channel.bound

inductive Proc : Type where
  | nil
  | one         (x : Channel) (P : Proc)
  | bot         (x : Channel) (P : Proc)
  | tensor      (x : Channel) (P : Proc)
  | parr        (x : Channel) (P : Proc)
  | cut         (P : Proc)
  | par         (P Q : Proc)
  | selectL     (x : Channel) (P : Proc)
  | selectR     (x : Channel) (P : Proc)
  | amp         (x : Channel) (P Q : Proc)
  | output      (x : Channel) (P : Proc) (A : Types)
  | input       (x : Channel) (P : Proc)
  | server      (x : Channel) (zs : Finset Channel) (P : Proc)
  | consume     (x : Channel) (P : Proc)
  | duplicate   (x : Channel) (P : Proc)
  | dispose     (x : Channel) (P : Proc)
  | link        (x y : Channel)
deriving DecidableEq, BEq
\end{lstlisting}

\subsection{Judgements}
\begin{lstlisting}[
  language=Lean4,
  caption={The main typing judgement. The judgement \texttt{Typing n P G} states that process \texttt{P} is well typed under hyperenvironment \texttt{G} at type-variable depth \texttt{n}.},
  label={lst:lean-typing-basic}
]
inductive Typing : Nat -> Proc -> HyperEnv -> Prop where
  | mix/-$_\texttt{0}$-/ {n : Nat} :
      Typing n /-$\nil$-/ /-$\emptyhyp$-/ 

  | mix {/-$\hyp{G}$-/ /-$\hyp{H}$-/ : HyperEnv} {P Q : Proc} {n : Nat} (hD : /-$\hyp{G}$-/.disjoint /-$\hyp{H}$-/) :
      Typing n P /-$\hyp{G}$-/ -> Typing n Q /-$\hyp{H}$-/ ->
      Typing n (P /-$\pp_\texttt{p}$-/ Q) (/-$\hyp{G}$-/ /-$\pp_\texttt{h}$-/ /-$\hyp{H}$-/)

  | one {P : Proc} {x : FPName} {n : Nat} :
      Typing n P /-$\emptyhyp$-/ ->
      Typing n (#x/-$\dbrackL$-//-$\dbrackR$-/.P) [[/-$\cht{\texttt{x}}{\texttt{1}}$-/]]

  | bot {/-$\Gamma$-/ : Env} {P : Proc} {x : FPName} {n : Nat} (hF : x /-$\notin$-/ /-$\Gamma$-/.names) :
      Typing n P [/-$\Gamma$-/] ->
      Typing n (#x/-$\dparenL$-//-$\dparenR$-/.P) [/-$\cht{\texttt{x}}{\bot} \texttt{::}\Gamma$-/]

  | cut {/-$\hyp{G}$-/ : HyperEnv} {/-$\Gamma$-/ /-$\Delta$-/ : Env} {P : Proc} {A : Types} {n : Nat} 
    (L : Finset FPName) :
      (/-$\forall$-/ x /-$\notin$-/ L, /-$\forall$-/ y /-$\notin$-/ L, x /-$\neq$-/ y ->
        Typing n (P/-$\dparenL$-/#x, #y/-$\dparenR$-/) (/-$\hyp{G}$-/ /-$\pp_\texttt{h}$-/ [/-$\cht{\texttt{x}}{\texttt{A}} \texttt{::} \Gamma$-/] /-$\pp_\texttt{h}$-/ [/-$\cht{\texttt{y}}{\dual{\texttt{A}}} \texttt{::} \Delta$-/])) ->
      Typing n (/-$\mathit{v}\dparenL$-//-$\bullet$-/,/-$\bullet$-//-$\dparenR$-/ P) (/-$\hyp{G}$-/ /-$\pp_\texttt{h}$-/ [/-$\Gamma$-/, /-$\Delta$-/])

The constructors in the above listing cover the full static \pill{} typing relation. The multiplicative binders \texttt{cut}, \texttt{tensor}, and \texttt{parr} use co-finite premises, while \texttt{exchange\_env} and \texttt{exchange\_hyper} internalise exchange for environments and hyperenvironments. These exchange constructors are used throughout the metatheory to compensate for the list-based representation of contexts.

\subsection{Transition Labels}\label{app:labels}
This appendix gives the full Lean definitions of the label and action datatypes discussed in \cref{subsec:semantics}, together with the associated free-name, introduced-name, and well-formedness functions used by the operational semantics.

\begin{lstlisting}[
    language={Lean4},
    caption={Definitions of transition labels, actions, and their associated free and introduced names.},
    label={lst:lean-label-definitions}
  ]
inductive Mu : Type
  | L
  | R
  | DISP
  | DUP
  | USE
deriving Repr, DecidableEq

inductive Act : Type
  | one       (x : FPName)
  | bot       (x : FPName)
  | tensor    (x y : FPName)
  | parr      (x y : FPName)
  | output    (x : FPName) (A : Types)
  | input     (x : FPName) (A : Types)
  | muBrack   (x : FPName) (/-$\mu$-/ : Mu)
  | muParen   (x : FPName) (/-$\mu$-/ : Mu)
deriving Repr, DecidableEq

inductive Lbl : Type
  | tau
  | link  (x y : FPName)
  | act   (p : Act)
  | par   (l l' : Act)
deriving Repr, DecidableEq

@[simp] def fNamesAct : Act -> Finset FPName -- free names
  | .one x | .bot x
  | .tensor x _ | .parr x _
  | .input x _ | .output x _
  | .muBrack x _  | .muParen x _ => {x}

@[simp] def iNamesAct : Act -> Finset FPName -- introduced names
  | .one _ | .bot _ | .input _ _ | .output _ _
  | Act.muBrack _ _ | Act.muParen _ _ => /-$\emptyset$-/
  | .tensor _ y | .parr _ y => {y}

@[simp] def Lbl.f : Lbl -> Finset FPName
  | .tau        => /-$\emptyset$-/
  | .link x y   => {x, y}
  | .act a      => fNamesAct a
  | .par l l'   => fNamesAct l /-$\cup$-/ fNamesAct l'

@[simp] def Lbl.i : Lbl -> Finset FPName
  | .tau        => /-$\emptyset$-/
  | .link _ _   => /-$\emptyset$-/
  | .act a      => iNamesAct a
  | .par l l'   => iNamesAct l /-$\cup$-/ iNamesAct l'

@[simp] def Lbl.WF : Lbl -> Prop
  | .tau => True
  | .link _ _ => True
  | .act _ => True
  | .par l l' => (iNamesAct l) /-$\cap$-/ (iNamesAct l') = /-$\emptyset$-/
\end{lstlisting}
